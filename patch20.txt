function expRenderAtts(){
  const el=document.getElementById('ex_attlist');if(!el)return;
  const pend=window.__EXFILES__||[];
  const pendHtml=pend.map(function(f,i){
    return '<div class="tot"><span>📎 '+esc(f.name)+' ('+(f.size/1048576).toFixed(2)+' MB)</span>'+
      '<span><span style="cursor:pointer;color:var(--bad)" onclick="expDelPend('+i+')">✕</span></span></div>';
  }).join('');
  const id=window.__EXID__;
  if(!id){el.innerHTML=pendHtml||'<span class="hint">No files attached yet. Attach the bill photo or PDF.</span>';return;}
  el.innerHTML=pendHtml+'<span class="hint">Loading saved files…</span>';
  fetch('/api/attachments?expenseId='+id).then(function(r){return r.ok?r.json():[];}).then(function(list){
    const el2=document.getElementById('ex_attlist');if(!el2)return;
    const p2=(window.__EXFILES__||[]).map(function(f,i){
      return '<div class="tot"><span>📎 '+esc(f.name)+' ('+(f.size/1048576).toFixed(2)+' MB)</span>'+
        '<span><span style="cursor:pointer;color:var(--bad)" onclick="expDelPend('+i+')">✕</span></span></div>';
    }).join('');
    el2.innerHTML=p2+(list.length?list.map(function(a){return expAttRow(a,id);}).join('')
      :'<span class="hint">No saved files for this expense yet.</span>');
  }).catch(function(){});
}
function expAddFiles(inp){
  window.__EXFILES__=window.__EXFILES__||[];
  for(const f of inp.files){
    if(f.size>5*1048576){toast(f.name+' is over 5 MB — skipped','bad');continue;}
    if(!/^image\\//.test(f.type)&&f.type!=='application/pdf'){toast(f.name+' — only images and PDFs can be attached','warn');continue;}
    window.__EXFILES__.push({name:f.name,mime:f.type||'application/octet-stream',size:f.size,file:f});
  }
  inp.value='';
  expRenderAtts();
}
function expDelPend(i){window.__EXFILES__=(window.__EXFILES__||[]).filter(function(x,n){return n!==i;});expRenderAtts();}
function expDelAtt(attId,expId){
  confirmBox('Delete this attachment?','The file will be removed from the server.',function(){
    fetch('/api/attachments?id='+attId,{method:'DELETE'}).then(function(){
      if(expId){expForm(expId);}else{closeModal();}
      toast('Attachment deleted','warn');});
  },'Delete',true);
}
function expUploadPending(expId){
  const pend=window.__EXFILES__||[];
  pend.forEach(function(f){
    if(!f.file)return;
    f.file.arrayBuffer().then(function(ab){
      return fetch('/api/attachments',{method:'POST',
        headers:{'content-type':f.mime,'x-expense-id':expId,'x-filename':encodeURIComponent(f.name)},body:ab});
    }).then(function(r){
      if(r.ok)toast('Attached '+f.name,'ok');
      else toast(f.name+' could not be attached','bad');
    }).catch(function(){toast(f.name+' will need to be attached again (offline)','warn');});
  });
  window.__EXFILES__=[];
}

function expPreview(id){
  modal('Preview','<div id="prevbox"><div class="hint">Loading…</div></div>',
    '<a class="btn ghost" href="/api/attachments/'+id+'?download=1" target="_blank">⤓ Download</a>'+
    '<button class="btn gold" onclick="closeModal()">Done</button>',{wide:1,nofocus:1});
  fetch('/api/attachments/'+id).then(function(r){
    if(!r.ok){document.getElementById('prevbox').innerHTML='<div class="alert e">Could not load the file.</div>';return null;}
    const mime=r.headers.get('content-type')||'';
    const el=document.getElementById('prevbox');
    if(/^image\\//.test(mime)){return r.blob().then(function(bl){
      const img=document.createElement('img');img.src=URL.createObjectURL(bl);
      img.style.cssText='max-width:100%';el.innerHTML='';el.appendChild(img);});}
    if(mime==='application/pdf'){return r.blob().then(function(bl){
      el.innerHTML='<iframe src="'+URL.createObjectURL(bl)+'" style="width:100%;height:70vh;border:1px solid var(--line);border-radius:8px"></iframe>';});}
    el.innerHTML='<div class="alert i"><span>●</span><div>This file type cannot be previewed here — use the Download button.</div></div>';
    return null;
  }).catch(function(){});
}
